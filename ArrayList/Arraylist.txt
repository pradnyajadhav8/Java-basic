import java.util.*;
class Sorting{
	public mstatic void main(String args[]){
		int[] arr={7,8,3,2,1};

		//bubble sort
		for(int i=0;i<arr.length;i++){//O(n^2)
			for(int j=0;j<arr.length;j++){
				if(arr[j]>arr[j+1]){
					//swap
					int temp= arr[j];
					arr[j]=arr[j+1];
					arr[j+1]=temp;
				}
			}
		}

		printarr(arr);
	}
}

import java.util.*;
class Sorting{
	public mstatic void main(String args[]){
		int[] arr={7,8,3,2,1};

		//Selection sort
		for(int i=0;i<arr.length;i++){//O(n^2)
			int smallest=i;
			for(int j=i+1;j<arr.length;j++){
				if(arr[smallest]>arr[j]){
					smallest=j;
				}
				int temp=arr[smallest];
				arr[smallest]=arr[i];
				arr[i]=temp;
			}
		}

		printarr(arr);
	}
}
